# NINJA GAIDEN - NEXT STEPS (run in fish, one block at a time)
# ============================================================
#
# 1) Git and the pinned engine + menu (only once)
#   Pins: engine 826fbbf (thread pinning, as Contra), menu c099b88 (tested on BOS).
#   The "git add" before the update stops it resetting the menu to another commit
#   (what happened in Contra).
cd ~/Projects/NinjaGaiden1Recomp
git init; git branch -m main
git submodule add -b dd2-240hz https://github.com/gabbycat4/nesrecomp.git nesrecomp
git submodule add https://github.com/mstan/recomp-ui.git recomp-ui
cd nesrecomp; git checkout 826fbbf; cd ../recomp-ui; git checkout c099b88; cd ..
git add nesrecomp recomp-ui
git submodule update --init --recursive
git ls-tree HEAD nesrecomp recomp-ui; git diff --cached --submodule=short nesrecomp recomp-ui
#   (before the first commit ls-tree shows nothing; after it, both lines must say 826fbbf... and c099b88...)
git add -A; git commit -m "Ninja Gaiden: project from the Contra template, pinned engine and menu"
git ls-tree HEAD nesrecomp recomp-ui

# 2) Headless build (the first build takes a while: it recompiles the game)
cd ~/Projects/NinjaGaiden1Recomp
cmake -S . -B build-headless -G Ninja -DCMAKE_BUILD_TYPE=Release -DNESRECOMP_HEADLESS=ON
cmake --build build-headless --parallel

# 3) First boot + probe: screenshots every 100 frames, so Claude can see the intro,
#   the title and where Act 1 starts (route.txt presses START at 300, 700, 1100, 1500)
cd ~/Projects/NinjaGaiden1Recomp
./build-headless/ng1recomp ninjagaiden.nes --frames 600 --screenshot boot.png
mkdir -p shots
./build-headless/ng1recomp ninjagaiden.nes --frames 3000 --input route.txt --shot-every 100 --screenshot shots/probe.png
#   Then tell Claude it's done (Claude reads the shots from the folder).
#   If step 2 or 3 errors, paste the error to Claude.
